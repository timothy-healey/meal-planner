# Foundation & Design System — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A runnable Expo Router app with a SQLite database, all design-system UI primitives, and a navigation skeleton — the shared foundation every other plan builds on.

**Architecture:** Expo Router provides file-based routing and tab navigation. A React context (`DatabaseProvider`) initialises SQLite once at startup and shares the connection to all hooks. UI primitives are atoms built on `constants/tokens.ts`; no component hard-codes a colour, size, or font name.

**Tech Stack:** Expo SDK (latest), React Native, TypeScript, Expo Router, `expo-sqlite` v2, `@expo-google-fonts/plus-jakarta-sans`, `react-native-reanimated`, `react-native-gesture-handler`, Jest + `@testing-library/react-native`

import React, { createContext, useContext, useEffect, useState } from 'react';
import * as SQLite from 'expo-sqlite';
import { runMigrations } from '../lib/db/migrations';

type DB = SQLite.SQLiteDatabase;
const DatabaseContext = createContext<DB | null>(null);

export function DatabaseProvider({ children }: { children: React.ReactNode }) {
  const [db, setDb] = useState<DB | null>(null);

  useEffect(() => {
    SQLite.openDatabaseAsync('meal-planner.db').then(async (database) => {
      await runMigrations(database);
      setDb(database);
    });
  }, []);

  if (!db) return null;
  return <DatabaseContext.Provider value={db}>{children}</DatabaseContext.Provider>;
}

export function useDb(): DB {
  const db = useContext(DatabaseContext);
  if (!db) throw new Error('useDb must be used inside DatabaseProvider');
  return db;
}
```

const MONTHS = ['Jan','Feb','Mar','Apr','May','Jun','Jul','Aug','Sep','Oct','Nov','Dec'];

export function formatCookTime(prep_minutes: number, cook_minutes: number): string {
  if (cook_minutes >= 60) {
    return `${Math.round((prep_minutes + cook_minutes) / 60)}h`;
  }
  return `${prep_minutes + cook_minutes} min`;
}

export function formatWeekOf(isoDate: string): string {
  const [, month, day] = isoDate.split('-').map(Number);
  return `${day} ${MONTHS[month - 1]}`;
}

export function formatPrice(amount: number): string {
  return `$${amount.toFixed(2)}`;
}

export function formatItemCount(checked: number, total: number): string {
  return `✓ ${checked} / ${total}`;
}
```

import { View } from 'react-native';

interface StepListProps {
  steps: string[];
  method?: string; // v1.0 fallback: single string
}

export function StepList({ steps, method }: StepListProps) {
  if (steps.length === 0 && method) {
    return <AppText color="textPrimary">{method}</AppText>;
  }

  return (
    <View style={{ gap: spacing[3] }}>
      {steps.map((step, index) => {
        const isLast = index === steps.length - 1;
        return (
          <View key={index} style={{ flexDirection: 'row', gap: spacing[3], alignItems: 'flex-start' }}>
            <View
              style={{
                width: 18,
                height: 18,
                borderRadius: 9,
                backgroundColor: isLast ? colors.orange : colors.green,
                alignItems: 'center',
                justifyContent: 'center',
                flexShrink: 0,
                marginTop: 2,
              }}
            >
              <AppText weight="bold" size="2xs" color="onGreen">{String(index + 1)}</AppText>
            </View>
            <AppText color="textPrimary" style={{ flex: 1 }}>{step}</AppText>
          </View>
        );
      })}
    </View>
  );
}
